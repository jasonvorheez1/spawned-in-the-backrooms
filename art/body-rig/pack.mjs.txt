import{NodeIO}from'@gltf-transform/core';import{ALL_EXTENSIONS}from'@gltf-transform/extensions';import{MeshoptDecoder,MeshoptEncoder}from'meshoptimizer';import{dedup,prune,sparse,meshopt}from'@gltf-transform/functions';import{readdir,mkdir,stat,writeFile}from'node:fs/promises';import assert from'node:assert/strict';
await Promise.all([MeshoptDecoder.ready,MeshoptEncoder.ready]);const io=new NodeIO().registerExtensions(ALL_EXTENSIONS).registerDependencies({'meshopt.decoder':MeshoptDecoder,'meshopt.encoder':MeshoptEncoder});const dir=(process.argv[2]??new URL('./',import.meta.url).pathname.replace(/^\/(C:)/,'$1')).replace(/[/\\]?$/,'/');await mkdir(dir+'packed',{recursive:true});let total=0,slots={};
for(const name of(await readdir(dir+'out')).filter(n=>n.endsWith('.glb'))){const doc=await io.read(dir+'out/'+name),old=await io.read(dir+name);const before=doc.getRoot().listNodes().filter(n=>n.getMesh()).map(n=>[n.getName(),n.getMesh().listPrimitives().length,n.getMesh().getExtras().targetNames]);const oldNodes=new Map(old.getRoot().listSkins().flatMap(s=>s.listJoints()).map(n=>[n.getName(),n]));
for(const skin of doc.getRoot().listSkins())for(const joint of skin.listJoints()){const prior=oldNodes.get(joint.getName());assert(prior,'lost joint '+joint.getName());const a=joint.getWorldMatrix(),b=prior.getWorldMatrix();const expected=[...b];if(!name.startsWith('body_')&&joint.getName().endsWith('ForeArm'))expected[12]+=(joint.getName().startsWith('Left')?1:-1)*.020;assert(Math.max(...a.map((v,i)=>Math.abs(v-expected[i])))<.0002,'unexpected bind transform '+name+' '+joint.getName())}
for(const n of old.getRoot().listNodes().filter(n=>n.getMesh())){const next=doc.getRoot().listNodes().find(q=>q.getName()===n.getName()&&q.getMesh());assert(next?.getMesh(),'lost mesh '+n.getName());assert.deepEqual(next.getMesh().getExtras().targetNames,n.getMesh().getExtras().targetNames,'morph loss '+name)}
await doc.transform(dedup(),prune({keepAttributes:true,keepLeaves:true}),sparse(),meshopt({encoder:MeshoptEncoder,level:'medium'}));await io.write(dir+'packed/'+name,doc);const check=await io.read(dir+'packed/'+name);const after=check.getRoot().listNodes().filter(n=>n.getMesh()).map(n=>[n.getName(),n.getMesh().listPrimitives().length,n.getMesh().getExtras().targetNames]);assert.deepEqual(after,before,'compression changed structure '+name);for(const mesh of check.getRoot().listMeshes())for(const p of mesh.listPrimitives()){const w=p.getAttribute('WEIGHTS_0');if(w){const a=w.getArray(),scale=w.getNormalized()?(w.getComponentType()===5121?255:65535):1;for(let i=0;i<a.length;i+=4)assert(Math.abs((a[i]+a[i+1]+a[i+2]+a[i+3])/scale-1)<.025,'unnormalized weights '+name)}}slots[name.slice(0,-4)]=[...new Set(check.getRoot().listMaterials().map(m=>m.getName()))];total+=(await stat(dir+'packed/'+name)).size;console.log('VERIFIED',name)}await writeFile(dir+'slots.json',JSON.stringify(slots,null,2));console.log('TOTAL_BYTES',total)
